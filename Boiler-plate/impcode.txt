function checkImageExists(url) {
    try {
        const xhr = new XMLHttpRequest();
        xhr.open('HEAD', url, false);  // Synchronous request
        xhr.send();
        
        if (xhr.status === 200) {
            const contentType = xhr.getResponseHeader('content-type');
            return contentType && contentType.startsWith('image');
        } else {
            return false;
        }
    } catch (error) {
        console.error('Error checking image existence:', error);
        return false; // Return false in case of any errors
    }
}

function slowNetworkResources() {
     console.log('calling slowNetworkResources function');
     window.performance.getEntriesByType('resource').forEach((entry) => {
         // Check if the resource was loaded slowly (e.g., over 5 seconds)
         if (entry.responseEnd - entry.startTime > 500) { // 5000ms = 5 seconds
             console.log(`Slow network detected for resource: ${entry.name}`);
             console.log(`Load time: ${(entry.responseEnd - entry.startTime) / 1000}s`);
         }
     });
 }

function captureNetworkLogs() {
     const originalAjax = $.ajax;
     console.log('captureNetworkLogs::ajax');

     $.ajax = function (options) {
         const startTime = Date.now();
         var method, url, payload;
         if (options) {
             method = options.method;
             url = options.url;
             payload = options.data;
         }
         console.log('Capturing request...');

         // Create an object to hold the data for the table
         const logData = {
             URL: url,
             Method: method,
             Payload: payload ? JSON.stringify(payload) : "No Payload",
             Status: "Pending",
             Duration: "Pending",
             Response: "Pending"
         };

         // Perform the AJAX request
         return originalAjax(options)
             .done((data, textStatus, jqXHR) => {
                 const duration = Date.now() - startTime;
                 logData.Status = jqXHR.status;
                 logData.Duration = duration + " ms";
                 logData.Response = JSON.stringify(data);

                 // Print the log data in table format
                 console.table([logData]);
             })
             .fail((jqXHR, textStatus, errorThrown) => {
                 logData.Status = "Error";
                 logData.Duration = Date.now() - startTime + " ms";
                 logData.Response = errorThrown;

                 // Print the error log in table format
                 console.table([logData]);
             });
     };
 }
 
 function serializeToObject(serializedData) {
    let obj = {};
    let pairs = serializedData.split('&');
    pairs.forEach(function (pair) {
        let [key, value] = pair.split('=');
        if (key) {
            obj[decodeURIComponent(key)] = decodeURIComponent(value || '');
        }
    });
    return obj;
}
